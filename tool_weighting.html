def render_dot_plot_svg(title: str, plot_id: str, isolate_scores: dict, cohort_mean: float) -> str:
    samples = sorted(isolate_scores.keys())
    n = len(samples)
    row_height = 24
    top_margin = 46
    bottom_margin = 40
    left_margin = 175
    plot_width = 450
    total_width = left_margin + plot_width + 65
    total_height = top_margin + (n * row_height) + bottom_margin

    def val_to_x(val: float) -> float:
        clipped = max(0.0, min(100.0, float(val)))
        return left_margin + (clipped / 100.0) * plot_width

    grid_lines = []
    tick_labels = []
    for tick in [0, 20, 40, 60, 80, 100]:
        x = val_to_x(tick)
        grid_lines.append(f'<line x1="{x:.1f}" y1="{top_margin}" x2="{x:.1f}" y2="{top_margin + n * row_height}" stroke="#e2e8f0" stroke-width="1" stroke-dasharray="2,2"/>')
        tick_labels.append(f'<text x="{x:.1f}" y="{top_margin + n * row_height + 18}" text-anchor="middle" font-size="11" fill="#64748b" font-family="sans-serif">{tick}</text>')

    mean_x = val_to_x(cohort_mean)
    mean_line = (
        f'<line x1="{mean_x:.1f}" y1="{top_margin - 8}" x2="{mean_x:.1f}" y2="{top_margin + n * row_height + 8}" '
        f'stroke="#0284c7" stroke-width="2.5" stroke-dasharray="4,3"/>\n'
        f'<rect x="{mean_x - 52:.1f}" y="{top_margin - 25}" width="104" height="18" rx="3" fill="#0284c7"/>\n'
        f'<text x="{mean_x:.1f}" y="{top_margin - 12}" text-anchor="middle" font-size="10.5" font-weight="bold" fill="#ffffff" font-family="sans-serif">Cohort Mean: {cohort_mean:.1f}</text>'
    )

    rows = []
    for idx, sample in enumerate(samples):
        y = top_margin + idx * row_height + (row_height / 2)
        score = float(isolate_scores.get(sample, 0.0))
        dot_x = val_to_x(score)
        rows.append(f'<line x1="{left_margin}" y1="{y:.1f}" x2="{left_margin + plot_width}" y2="{y:.1f}" stroke="#f8fafc" stroke-width="1"/>')
        rows.append(f'<text x="{left_margin - 10}" y="{y + 4:.1f}" text-anchor="end" font-size="11" fill="#334155" font-family="monospace">{sample}</text>')
        # Dual visual encoding: filled circle + white ring (colour is not the only carrier)
        rows.append(
            f'<circle cx="{dot_x:.1f}" cy="{y:.1f}" r="6" fill="#0f766e" stroke="#134e4a" stroke-width="1.5"/>\n'
            f'<circle cx="{dot_x:.1f}" cy="{y:.1f}" r="2" fill="#ffffff"/>'
        )
        text_x = dot_x + 10 if dot_x < (left_margin + plot_width - 32) else dot_x - 10
        text_anchor = "start" if dot_x < (left_margin + plot_width - 32) else "end"
        rows.append(f'<text x="{text_x:.1f}" y="{y + 3.5:.1f}" text-anchor="{text_anchor}" font-size="10" font-weight="600" fill="#0f766e" font-family="sans-serif">{score:.1f}</text>')

    return f"""
    <svg id="{plot_id}" width="100%" height="{total_height}" viewBox="0 0 {total_width} {total_height}" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="{title} per-isolate dot plot">
        <title>{title} Dot Plot</title>
        <desc>{title} scores for 13 reference bacterial isolates with cohort mean at {cohort_mean:.1f}. No population confidence intervals are displayed as this is the complete benchmark cohort.</desc>
        <text x="{left_margin}" y="18" font-size="13" font-weight="bold" fill="#1e293b" font-family="sans-serif">{title}</text>
        <text x="{left_margin}" y="32" font-size="10" fill="#64748b" font-family="sans-serif">Fixed 0–100 criterion scale | Reference cohort (no population confidence intervals)</text>
        {''.join(grid_lines)}
        {''.join(tick_labels)}
        {''.join(rows)}
        {mean_line}
    </svg>
    """

def calculate_scenario_rankings(
    df: pd.DataFrame,
    model: str,
    depth: str,
    weight_accuracy: float,
    weight_contiguity: float,
    weight_residual: float,
    weight_replicon: float,
    gate_complete: bool = False,
    gate_zero_hits: bool = False,
):
    subset = df[(df["model"] == model) & (df["depth"] == depth)]
    w_acc = float(weight_accuracy) / 100.0
    w_cont = float(weight_contiguity) / 100.0
    w_res = float(weight_residual) / 100.0
    w_rep = float(weight_replicon) / 100.0

    eligible_records = []
    excluded_records = []
    insufficient_records = []

    combos_present = set(subset["combo"].unique()) if not subset.empty else set()
    all_combos = sorted(EXPECTED_BENCHMARK_COMBINATIONS.union(combos_present))

    for combo in all_combos:
        combo_name = str(combo)
        isolate_metrics = subset[subset["combo"] == combo]
        samples_present = set(isolate_metrics["sample"].unique())
        missing_samples = EXPECTED_BENCHMARK_SAMPLES - samples_present
        if missing_samples:
            insufficient_records.append({
                "combo": combo_name,
                "overall_score": 0.0,
                "display_score": None,
                "rank": None,
                "is_eligible": False,
                "is_insufficient_data": True,
                "ineligible_reason": (
                    f"insufficient benchmark data: missing {len(missing_samples)} isolate(s): "
                    f"{sorted(missing_samples)}"
                ),
                "reasons": [
                    f"insufficient benchmark data: missing {len(missing_samples)} isolate(s): "
                    f"{sorted(missing_samples)}"
                ],
                "warnings": ["Insufficient benchmark data"],
                "raw_group": isolate_metrics,
            })
            continue

        iso_cont = np.maximum(0.0, 1.0 - np.abs(isolate_metrics["auNGA_ratio"] - 1.0)) * 100.0
        iso_acc = np.clip(1.0 - (isolate_metrics["Mismatches per 100kbp"] + isolate_metrics["Indels per 100kbp"]) / 10.0, 0.0, 1.0) * 100.0

        mean_cont = float(np.mean(iso_cont))
        mean_acc = float(np.mean(iso_acc))
        clean_pct = float(np.mean(isolate_metrics["contamination_count"] == 0) * 100.0)
        total_residual_hits = int(isolate_metrics["contamination_count"].sum())
        clean_isolates = int((isolate_metrics["contamination_count"] == 0).sum())
        affected_residual_isolates = int((isolate_metrics["contamination_count"] > 0).sum())
        total_missed_replicons = int(isolate_metrics["total_missed"].sum())
        full_missed = int(isolate_metrics["full_missed"].sum())
        partial_missed = int(isolate_metrics["partial_missed"].sum())
        affected_replicon_isolates = int((isolate_metrics["total_missed"] > 0).sum())
        score_rep = float(max(0.0, 1.0 - total_missed_replicons / 3.0) * 100.0)

        mean_error_rate = float(np.mean(isolate_metrics["Mismatches per 100kbp"] + isolate_metrics["Indels per 100kbp"]))
        mean_mismatches = float(np.mean(isolate_metrics["Mismatches per 100kbp"]))
        mean_indels = float(np.mean(isolate_metrics["Indels per 100kbp"]))
        mean_aunga = float(np.mean(isolate_metrics["auNGA_ratio"]))

        # Supporting diagnostics
        dup_series = isolate_metrics["Duplication_ratio"] if "Duplication_ratio" in isolate_metrics.columns else pd.Series([1.0] * len(isolate_metrics))
        mean_duplication = float(np.mean(dup_series))
        mis_series = isolate_metrics["misassemblies"] if "misassemblies" in isolate_metrics.columns else pd.Series([0] * len(isolate_metrics))
        total_misassemblies = int(np.sum(mis_series))

        overall_score = (w_acc * mean_acc + w_cont * mean_cont + w_res * clean_pct + w_rep * score_rep)

        # Check replicon completeness (>=95% coverage on all expected replicons, rejecting empty strings)
        all_replicons_complete = True
        for cov_str in isolate_metrics["all_contigs_coverage"]:
            text = str(cov_str).strip()
            if not text:
                all_replicons_complete = False
                break
            matches = _REPLICON_COV_RE.findall(text)
            if not matches or any(float(m) < 95.0 for m in matches):
                all_replicons_complete = False
                break

        is_eligible = True
        reasons = []

        if gate_complete and not all_replicons_complete:
            is_eligible = False
            reasons.append("fails complete-recovery gate (<95% coverage on at least one replicon)")

        if gate_zero_hits and total_residual_hits > 0:
            is_eligible = False
            reasons.append(f"fails zero-residual-hits gate ({total_residual_hits} residual hit(s) detected)")

        # Concrete warnings
        warnings = []
        if total_residual_hits > 0:
            warnings.append(f"{total_residual_hits} residual adapter/barcode hit(s) across {affected_residual_isolates} isolate(s)")
        if total_missed_replicons > 0:
            warnings.append(f"{total_missed_replicons} missed or severely incomplete replicon(s) across {affected_replicon_isolates} isolate(s) ({full_missed} full, {partial_missed} partial)")

        min_acc_val = float(np.min(iso_acc))
        if (mean_acc - min_acc_val) >= 25.0:
            warnings.append(f"Variable sequence accuracy across isolates: lowest isolate score is {min_acc_val:.1f} (cohort mean {mean_acc:.1f})")

        min_cont_val = float(np.min(iso_cont))
        if (mean_cont - min_cont_val) >= 25.0:
            warnings.append(f"Variable contiguity across isolates: lowest isolate score is {min_cont_val:.1f} (cohort mean {mean_cont:.1f})")

        sample_names = list(isolate_metrics["sample"])
        iso_acc_dict = {str(s): float(v) for s, v in zip(sample_names, iso_acc)}
        iso_cont_dict = {str(s): float(v) for s, v in zip(sample_names, iso_cont)}

        item = {
            "combo": combo_name,
            "overall_score": overall_score,
            "display_score": round(overall_score, 1),
            "rank": None,
            "is_eligible": is_eligible,
            "is_insufficient_data": False,
            "contiguity": round(mean_cont, 1),
            "accuracy": round(mean_acc, 1),
            "residual": round(clean_pct, 1),
            "replicon": round(score_rep, 1),
            "score_contiguity": mean_cont,
            "score_accuracy": mean_acc,
            "score_residual": clean_pct,
            "score_replicon": score_rep,
            "mean_error_rate": mean_error_rate,
            "mean_mismatches": mean_mismatches,
            "mean_indels": mean_indels,
            "mean_aunga": mean_aunga,
            "mean_duplication_ratio": mean_duplication,
            "total_misassemblies": total_misassemblies,
            "isolate_acc_scores": iso_acc_dict,
            "isolate_cont_scores": iso_cont_dict,
            "t_hits": total_residual_hits,
            "clean_isolates": clean_isolates,
            "affected_residual_isolates": affected_residual_isolates,
            "t_miss": total_missed_replicons,
            "full_missed": full_missed,
            "partial_missed": partial_missed,
            "affected_replicon_isolates": affected_replicon_isolates,
            "reasons": reasons,
            "ineligible_reason": "; ".join(reasons) if reasons else None,
            "warnings": warnings,
            "is_near_tie": False,
            "raw_group": isolate_metrics,
        }

        if is_eligible:
            eligible_records.append(item)
        else:
            excluded_records.append(item)

    # Sort eligible by full precision overall_score descending, then combo
    eligible_records.sort(key=lambda r: (-r["overall_score"], r["combo"]))

    # Competition ranking
    for idx, rec in enumerate(eligible_records):
        if idx == 0:
            rec["rank"] = 1
        else:
            prev = eligible_records[idx - 1]
            if abs(rec["overall_score"] - prev["overall_score"]) < 1e-9:
                rec["rank"] = prev["rank"]
            else:
                rec["rank"] = idx + 1

    # Near tie calculation
    for idx in range(len(eligible_records) - 1):
        if abs(eligible_records[idx]["overall_score"] - eligible_records[idx + 1]["overall_score"]) < 1.0:
            eligible_records[idx]["is_near_tie"] = True
            eligible_records[idx + 1]["is_near_tie"] = True

    excluded_records.sort(key=lambda r: (-r["overall_score"], r["combo"]))
    insufficient_records.sort(key=lambda r: r["combo"])

    return eligible_records, excluded_records, insufficient_records

def generate_dashboard_csv_export(
    df: pd.DataFrame,
    model: str,
    depth: str,
    preset: str,
    weight_accuracy: float,
    weight_contiguity: float,
    weight_residual: float,
    weight_replicon: float,
    gate_complete: bool = False,
    gate_zero_hits: bool = False,
    config=None,
) -> str:
    if not model or not depth:
        raise ValueError("Cannot generate CSV export: Scenario must be selected (basecalling model and depth).")

    w_acc = float(weight_accuracy)
    w_cont = float(weight_contiguity)
    w_res = float(weight_residual)
    w_rep = float(weight_replicon)
    if w_acc < 0 or w_cont < 0 or w_res < 0 or w_rep < 0:
        raise ValueError("Cannot generate CSV export: Weights must be non-negative.")
    total = w_acc + w_cont + w_res + w_rep
    if abs(total - 100.0) > 1e-4 or total == 0:
        raise ValueError("Cannot generate CSV export: Weights must total exactly 100%.")

    eligible, excluded, insufficient = calculate_scenario_rankings(
        df=df,
        model=model,
        depth=depth,
        weight_accuracy=w_acc,
        weight_contiguity=w_cont,
        weight_residual=w_res,
        weight_replicon=w_rep,
        gate_complete=gate_complete,
        gate_zero_hits=gate_zero_hits,
    )

    csv_bytes = df.to_csv(index=False).encode("utf-8")
    content_hash = hashlib.sha256(csv_bytes).hexdigest()
    now_iso = datetime.now(timezone.utc).isoformat()

    scoring_version = getattr(config, "scoring_version", "1.0") if config else "1.0"
    source_commit = getattr(config, "pinned_commit", "4d6b8cb1d482e5066f9ca575ebd3b67af4a32562") if config else "4d6b8cb1d482e5066f9ca575ebd3b67af4a32562"

    def _build_row(r, rank_val, is_elig, is_insuff, inelig_reason, is_nt, nt_label, disp_score):
        return {
            "rank": rank_val,
            "combo": r["combo"],
            "overall_score": r["overall_score"] if not is_insuff else 0.0,
            "display_score": disp_score,
            "is_eligible": is_elig,
            "is_insufficient_data": is_insuff,
            "ineligible_reason": inelig_reason,
            "is_near_tie": is_nt,
            "near_tie_label": nt_label,
            "score_contiguity": r["score_contiguity"] if not is_insuff else 0.0,
            "score_accuracy": r["score_accuracy"] if not is_insuff else 0.0,
            "score_residual": r["score_residual"] if not is_insuff else 0.0,
            "score_replicon": r["score_replicon"] if not is_insuff else 0.0,
            "mean_auNGA_ratio": r["mean_aunga"] if not is_insuff else 0.0,
            "mean_error_rate": r["mean_error_rate"] if not is_insuff else 0.0,
            "mean_mismatches": r["mean_mismatches"] if not is_insuff else 0.0,
            "mean_indels": r["mean_indels"] if not is_insuff else 0.0,
            "residual_total_hits": r["t_hits"] if not is_insuff else 0,
            "residual_clean_isolates": r["clean_isolates"] if not is_insuff else 0,
            "residual_affected_isolates": r["affected_residual_isolates"] if not is_insuff else 0,
            "replicon_total_missed": r["t_miss"] if not is_insuff else 0,
            "replicon_full_missed": r["full_missed"] if not is_insuff else 0,
            "replicon_partial_missed": r["partial_missed"] if not is_insuff else 0,
            "replicon_affected_isolates": r["affected_replicon_isolates"] if not is_insuff else 0,
            "mean_duplication_ratio": r["mean_duplication_ratio"] if not is_insuff else 0.0,
            "total_misassemblies": r["total_misassemblies"] if not is_insuff else 0,
            "warnings": "; ".join(r["warnings"]),
            "preset": preset,
            "scenario_model": model,
            "scenario_depth": depth,
            "weight_accuracy": w_acc,
            "weight_contiguity": w_cont,
            "weight_residual": w_res,
            "weight_replicon": w_rep,
            "gate_complete_recovery": bool(gate_complete),
            "gate_zero_residual_hits": bool(gate_zero_hits),
            "scoring_version": scoring_version,
            "source_data_commit": source_commit,
            "source_data_hash": content_hash,
            "generated_at": now_iso,
        }

    rows = []
    # 1. Eligible combinations in displayed ranking order
    for r in eligible:
        rows.append(_build_row(
            r,
            rank_val=r["rank"] if r["rank"] is not None else "",
            is_elig=True,
            is_insuff=False,
            inelig_reason="",
            is_nt=r["is_near_tie"],
            nt_label="similar overall scores" if r["is_near_tie"] else "",
            disp_score=r["display_score"] if r["display_score"] is not None else "",
        ))

    # 2. Excluded combinations in displayed order
    for r in excluded:
        rows.append(_build_row(
            r,
            rank_val="",
            is_elig=False,
            is_insuff=False,
            inelig_reason="; ".join(r["reasons"]),
            is_nt=r["is_near_tie"],
            nt_label="similar overall scores" if r["is_near_tie"] else "",
            disp_score=r["display_score"] if r["display_score"] is not None else "",
        ))

    # 3. Insufficient data combinations in displayed order
    for r in insufficient:
        rows.append(_build_row(
            r,
            rank_val="",
            is_elig=False,
            is_insuff=True,
            inelig_reason="; ".join(r["reasons"]),
            is_nt=False,
            nt_label="",
            disp_score="",
        ))

    export_df = pd.DataFrame(rows)
    buf = io.StringIO()
    export_df.to_csv(buf, index=False)
    return buf.getvalue()

    def _matches_config(current, expected):
        return (
            abs(current[0] - expected[0]) < 1e-4
            and abs(current[1] - expected[1]) < 1e-4
            and abs(current[2] - expected[2]) < 1e-4
            and abs(current[3] - expected[3]) < 1e-4
            and current[4] == expected[4]
            and current[5] == expected[5]

    def _detect_manual_edits():
        preset_name = input.preset_select()
        if preset_name not in PRESET_DEFINITIONS:
            return
        weight_accuracy = input.w_acc()
        weight_contiguity = input.w_cont()
        weight_residual = input.w_res()
        weight_replicon = input.w_rep()
        gate_complete_checked = input.gate_complete()
        gate_zero_hits_checked = input.gate_zero_hits()
        if None in (
            weight_accuracy,
            weight_contiguity,
            weight_residual,
            weight_replicon,
            gate_complete_checked,
            gate_zero_hits_checked,
        ):
            return
        current_config = (
            float(weight_accuracy),
            float(weight_contiguity),
            float(weight_residual),
            float(weight_replicon),
            bool(gate_complete_checked),
            bool(gate_zero_hits_checked),
        )
        in_flight_preset = target_preset.get()
        if in_flight_preset is not None:
            if in_flight_preset != preset_name:
                target_preset.set(None)
                return
            expected_config = PRESET_DEFINITIONS[in_flight_preset]
            if _matches_config(current_config, expected_config):
                target_preset.set(None)
            return

        active_preset_config = PRESET_DEFINITIONS[preset_name]
        if not _matches_config(current_config, active_preset_config):
